01 - Melyik karakter vagyok?

int x = 10; -> változó
int* p = &x; -> pointer
void f(int x) { x=99; } -> érték szerinti paraméterátadás
void f(int& x) { x=99; } -> referencia szerinti paraméterátadás
int* p = nullptr; -> null pointer
{ int n=5; } -> lokális változó (scope)
template<typename T> T add(T a, T b) -> template (sablon)
return n * f(n-1); -> rekurzió
